cmake_minimum_required(VERSION 4.3.1)

set(_aegra_version_h "${CMAKE_CURRENT_SOURCE_DIR}/include/aegra_version.h")
if(NOT EXISTS "${_aegra_version_h}")
    message(FATAL_ERROR "Missing ${_aegra_version_h}")
endif()
file(STRINGS "${_aegra_version_h}" _aegra_version_lines REGEX "^#define AEGRI_VERSION_(MAJOR|MINOR|PATCH) ")
foreach(_aegra_line IN LISTS _aegra_version_lines)
    if(_aegra_line MATCHES "^#define AEGRI_VERSION_MAJOR ([0-9]+)")
        set(_aegra_ver_major "${CMAKE_MATCH_1}")
    elseif(_aegra_line MATCHES "^#define AEGRI_VERSION_MINOR ([0-9]+)")
        set(_aegra_ver_minor "${CMAKE_MATCH_1}")
    elseif(_aegra_line MATCHES "^#define AEGRI_VERSION_PATCH ([0-9]+)")
        set(_aegra_ver_patch "${CMAKE_MATCH_1}")
    endif()
endforeach()
if(NOT DEFINED _aegra_ver_major OR NOT DEFINED _aegra_ver_minor OR NOT DEFINED _aegra_ver_patch)
    message(FATAL_ERROR "Could not parse version from ${_aegra_version_h}")
endif()

project(Aegra VERSION ${_aegra_ver_major}.${_aegra_ver_minor}.${_aegra_ver_patch} LANGUAGES CXX)

option(AEGRA_WARNINGS_AS_ERRORS "Treat Aegra compiler warnings as errors" ON)
option(AEGRA_BUILD_DESKTOP "Build the Qt desktop client" OFF)
set(AEGRA_QT_ROOT "" CACHE PATH "Qt installation prefix for the desktop client")

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(AegraTarget)
