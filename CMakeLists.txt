add_subdirectory(src/adapters/sqlite)
if(WIN32)

    add_subdirectory(src/adapters/windows_disk)
    add_subdirectory(src/adapters/windows_filesystem)
    add_subdirectory(src/adapters/windows_ipc)
    add_subdirectory(src/adapters/windows_system)
    add_subdirectory(src/adapters/windows_vss)
    add_subdirectory(src/adapters/windows_process)
    add_subdirectory(src/adapters/dokan)
    add_subdirectory(src/adapters/ntfs)
    add_subdirectory(src/apps/service)
    add_subdirectory(src/apps/worker)
    add_subdirectory(src/apps/shell_extension)
    if(AEGRA_BUILD_DESKTOP)
        if(AEGRA_QT_ROOT)
            list(PREPEND CMAKE_PREFIX_PATH "${AEGRA_QT_ROOT}")
